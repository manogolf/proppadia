#!/bin/zsh

SCRIPT_MODE="run"

MLB_CHECK_TOMORROW=0

  case "$1" in
    --check)
      SCRIPT_MODE="check"
      shift
      ;;
    --date)
      if [[ "${2:-}" == "" ]]; then
        echo "Usage: $0 [--check] [--date YYYY-MM-DD] [--tomorrow]" >&2
        exit 2
      fi
      MLB_CHECK_DATE_OVERRIDE="$2"
      shift 2
      ;;
    --tomorrow)
      MLB_CHECK_TOMORROW=1
      shift
      ;;
    *)
      echo "Usage: $0 [--check] [--date YYYY-MM-DD] [--tomorrow]" >&2
      exit 2
      ;;
  esac
done

  echo "--date/--tomorrow are only supported with --check" >&2
  exit 2
fi
if [[ "$MLB_CHECK_DATE_OVERRIDE" != "" && "$MLB_CHECK_TOMORROW" == "1" ]]; then
  echo "--date and --tomorrow are mutually exclusive" >&2
  exit 2
fi

fi

from pathlib import Path

def env(name: str, default: str = "") -> str:
    return os.environ.get(name, default)

def as_bool(value: str) -> bool:
    return str(value).strip().lower() in {"1", "true", "yes", "y"}

def as_int(value: str) -> int:
    try:
        return int(value)
    except Exception:
        return 0

def file_info(path_value: str) -> dict:
    path = Path(path_value) if path_value else Path("")
    exists = bool(path_value) and path.exists()
    return {
        "path": path_value,
        "exists": exists,
        "size": path.stat().st_size if exists and path.is_file() else 0,
    }

def load_json(path_value: str) -> dict:
    if not path_value:
        return {}
    path = Path(path_value)
    if not path.exists():
        return {}
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        return {"_load_error": f"{type(exc).__name__}: {exc}"}
